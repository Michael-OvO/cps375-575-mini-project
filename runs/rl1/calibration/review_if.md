    def strip_bits(s: str) -> str | None:
        digits = re.sub(r"[\W\s]", "", s)
        if not digits.isdigit() or len(digits) != 10:
            return None
        return digits

    parts = [
        [s.strip(), result]
        for s, result in map(

        if result is not None
    ]

    return [

    ]
